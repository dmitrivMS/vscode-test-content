\section{The Fourier Transform}
The Fourier transform of a function $f(t)$ is defined as:
\begin{equation}
    \hat{f}(\omega) = \int_{-\infty}^{\infty} f(t) \, e^{-i\omega t} \, dt
\end{equation}

The inverse transform recovers $f(t)$ from $\hat{f}(\omega)$:
\begin{equation}
    f(t) = \frac{1}{2\pi} \int_{-\infty}^{\infty} \hat{f}(\omega) \, e^{i\omega t} \, d\omega
\end{equation}

\section{Properties}
\begin{itemize}
    \item \textbf{Linearity:} $\mathcal{F}\{af + bg\} = a\hat{f} + b\hat{g}$
    \item \textbf{Time shift:} $\mathcal{F}\{f(t - t_0)\} = e^{-i\omega t_0}\hat{f}(\omega)$
\end{itemize}
